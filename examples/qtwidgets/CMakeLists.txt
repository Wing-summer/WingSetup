cmake_minimum_required(VERSION 3.21.1)
project(WingSetupWidgets LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_WINDOWS_EXPORT_ALL_SYMBOLS ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets LinguistTools)
add_library(private_data SHARED private_data.cpp)
add_library(demo_plugin MODULE plugin.cpp)
target_link_libraries(demo_plugin PRIVATE Qt6::Core private_data)
set_target_properties(demo_plugin PROPERTIES PREFIX "" SUFFIX ".wingplg"
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugin"
    INSTALL_RPATH "$ORIGIN/../../lib")
add_executable(widgets-demo main.cpp)
target_link_libraries(widgets-demo PRIVATE Qt6::Widgets)
qt_add_translation(QM_FILES demo_zh_CN.ts)
add_custom_target(translations ALL DEPENDS ${QM_FILES})
if(APPLE)
    set_target_properties(widgets-demo PROPERTIES MACOSX_BUNDLE TRUE
        MACOSX_BUNDLE_GUI_IDENTIFIER org.example.Widgets)
    set(APP_LOCATION "widgets-demo.app/Contents/MacOS")
    set(LIB_LOCATION "widgets-demo.app/Contents/Frameworks")
    set(RESOURCE_LOCATION "widgets-demo.app/Contents/Resources")
    set(PLUGIN_LOCATION "widgets-demo.app/Contents/PlugIns")
    set_target_properties(demo_plugin PROPERTIES INSTALL_RPATH "@loader_path/../Frameworks")
    install(TARGETS widgets-demo BUNDLE DESTINATION .)
elseif(WIN32)
    set(APP_LOCATION ".")
    set(LIB_LOCATION ".")
    set(RESOURCE_LOCATION ".")
    set(PLUGIN_LOCATION "plugin")
    install(TARGETS widgets-demo RUNTIME DESTINATION .)
else()
    set(APP_LOCATION "bin")
    set(LIB_LOCATION "lib")
    set(RESOURCE_LOCATION "bin")
    set(PLUGIN_LOCATION "bin/plugin")
    install(TARGETS widgets-demo RUNTIME DESTINATION bin)
endif()
install(TARGETS demo_plugin LIBRARY DESTINATION "${PLUGIN_LOCATION}" RUNTIME DESTINATION "${PLUGIN_LOCATION}")
install(TARGETS private_data LIBRARY DESTINATION "${LIB_LOCATION}" RUNTIME DESTINATION "${LIB_LOCATION}")
install(FILES ${QM_FILES} DESTINATION "${RESOURCE_LOCATION}/lang")
install(DIRECTORY assets/scripts assets/theme DESTINATION "${RESOURCE_LOCATION}")
install(FILES assets/settings.json DESTINATION "${RESOURCE_LOCATION}")
install(FILES org.example.Widgets.desktop DESTINATION share/applications)
install(FILES org.example.Widgets.svg DESTINATION share/icons/hicolor/scalable/apps)
